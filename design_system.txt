/
├── (GuestLayout)
│   ├── index                                 -> Home / landing
│   ├── doctors                               -> Doctor listing
│   ├── doctors/:doctorId                     -> Doctor detail profile
│   ├── specialties                           -> Specialty listing
│   ├── specialties/:specialtySlug            -> Specialty detail + doctors
│   ├── hospitals                             -> Hospital/clinic listing
│   ├── hospitals/:hospitalId                 -> Hospital detail + doctors
│   ├── news                                  -> News listing
│   ├── news/:newsId                          -> News detail
│   ├── services                              -> Service listing
│   ├── services/:serviceId                   -> Service detail
│   ├── search                                -> Unified search results
│   ├── about                                 -> Platform/about page
│   └── contact                               -> Contact/support

/auth
├── (AuthLayout + GuestOnlyGuard)
│   ├── login                                 -> Sign in
│   ├── register                              -> Sign up
│   ├── forgot-password                       -> Request reset link
│   ├── reset-password/:token                 -> Reset password
│   ├── verify-email/:token                   -> Email verification
│   └── oauth/callback                        -> Social login callback

/app
├── (ProtectedGuard + AppLayout)
│   ├── user
│   │   ├── (RoleGuard: patient/user)
│   │   ├── dashboard                          -> Trang tổng quan của người dùng / bệnh nhân
│   │   ├── booking/:doctorId                  -> Điền form và xác nhận đặt lịch
│   │   ├── appointments                       -> Bệnh nhân được xem lịch hẹn của mình (quá khứ, sắp tới)
│   │   ├── medical-records                    -> Bệnh nhân được xem hồ sơ bệnh án của mình
│   │   └── settings                           -> Trang cài đặt của bác sĩ
│   ├── doctor
│   │   ├── (RoleGuard: doctor)
│   │   ├── dashboard                          -> Trang tổng quan của bác sĩ 
│   │   ├── schedule                           -> Bác sĩ chỉ được xem lịch làm việc, hoặc xin nghỉ phép
│   │   ├── appointments                       -> Bác sĩ chỉ được xem lịch hẹn với bệnh nhân, và điền form khám bệnh
│   │   ├── patients                           -> Bác sĩ chỉ được xem thông tin bệnh nhân mình đã khám
│   │   └── settings                           -> Trang cài đặt của bác sĩ
│   └── admin
│       ├── (RoleGuard: admin)
│       ├── dashboard                          -> Trang tổng quan của admin
│       ├── doctors                            -> Admin toàn quyền quản lý bác sĩ
│       ├── schedules                          -> Admin toàn quyền sắp lịch trình cho bác sĩ
│       ├── appointments                       -> Admin toàn quyền quản lý các lịch hẹn của các bác sĩ
│       ├── patients                           -> Admin được xem toàn bộ bệnh nhân đã đặt lịch khám trên trang web
│       ├── specialties                        -> Admin toàn quyền quản lý các chuyên khoa
│       ├── news                               -> Admin toàn quyền quản lý các bài viết
│       ├── services                           -> Admin toàn quyền quản lý các dịch vụ
│       ├── hospitals                          -> Admin toàn quyền quản lý phòng khám / bệnh viện hợp tác
│       ├── users                              -> Admin toàn quyền quản lý tài khoản
│       ├── reports                            -> Admin có thể xem báo cáo doanh số
│       └── settings                           -> Trang cài đặt của admin

/system
├── 403                                        -> Unauthorized
├── 404                                        -> Not found
└── 500                                        -> Error fallback

*                                              -> Catch-all redirect to /system/404
